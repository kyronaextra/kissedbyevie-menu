## control.py — Docker control without the CLI

```python
"""Drive Docker over the unix socket. No docker binary needed."""
from __future__ import annotations

import os

import httpx

PROJECT = os.environ.get("COMPOSE_PROJECT", "evieprofiler")
ALLOWED = {"api", "listener", "backfill"}

_client = httpx.AsyncClient(
    transport=httpx.AsyncHTTPTransport(uds="/var/run/docker.sock"),
    base_url="http://docker", timeout=30,
)


def cname(service: str) -> str:
    return f"{PROJECT}-{service}-1"


async def list_services() -> list[dict]:
    r = await _client.get("/containers/json", params={"all": "1"})
    r.raise_for_status()
    out = []
    for c in r.json():
        names = [n.lstrip("/") for n in c.get("Names", [])]
        if not names or not names[0].startswith(PROJECT):
            continue
        name = names[0]
        out.append({
            "service": name.replace(f"{PROJECT}-", "").rsplit("-", 1)[0],
            "container": name,
            "state": c.get("State"),
            "status": c.get("Status"),
            "id": c.get("Id", "")[:12],
        })
    return sorted(out, key=lambda x: x["service"])


async def action(service: str, verb: str) -> dict:
    if service not in ALLOWED:
        raise ValueError(f"unknown service {service}")
    if verb not in {"start", "stop", "restart"}:
        raise ValueError(f"unknown verb {verb}")
    r = await _client.post(f"/containers/{cname(service)}/{verb}", params={"t": "10"})
    if r.status_code == 304:
        return {"ok": True, "service": service, "verb": verb, "noop": True}
    r.raise_for_status()
    return {"ok": True, "service": service, "verb": verb}


def _demux(b: bytes) -> str:
    out, i = [], 0
    while i + 8 <= len(b):
        size = int.from_bytes(b[i + 4:i + 8], "big")
        out.append(b[i + 8:i + 8 + size].decode("utf-8", "replace"))
        i += 8 + size
    return "".join(out) or b.decode("utf-8", "replace")


async def logs(service: str, tail: int = 200) -> str:
    r = await _client.get(
        f"/containers/{cname(service)}/logs",
        params={"stdout": "1", "stderr": "1", "tail": str(tail), "timestamps": "0"},
    )
    r.raise_for_status()
    return _demux(r.content)[-20000:]
```

## settings.py — shared pause flag

```python
"""Tiny k/v table so the panel and the listener can agree on state."""
import time

import db

_cache: dict[str, tuple[float, str]] = {}


def _init() -> None:
    db.conn().execute(
        "CREATE TABLE IF NOT EXISTS settings (k TEXT PRIMARY KEY, v TEXT, updated_at INTEGER)"
    )


def set_(k: str, v: str) -> None:
    _init()
    with db.tx() as c:
        c.execute(
            """INSERT INTO settings (k, v, updated_at) VALUES (?,?,?)
               ON CONFLICT(k) DO UPDATE SET v=excluded.v, updated_at=excluded.updated_at""",
            (k, v, int(time.time())),
        )
    _cache.pop(k, None)


def get(k: str, default: str = "") -> str:
    hit = _cache.get(k)
    if hit and time.time() - hit[0] < 5:
        return hit[1]
    _init()
    row = db.conn().execute("SELECT v FROM settings WHERE k=?", (k,)).fetchone()
    val = row["v"] if row else default
    _cache[k] = (time.time(), val)
    return val


def paused() -> bool:
    return get("ingest_paused", "0") == "1"
```

```python
import hmac
import os

from fastapi import Header, HTTPException

MASTER = os.environ.get("PROFILER_TOKEN", "")
CONTROL = os.environ.get("PROFILER_CONTROL_TOKEN", "")


def _eq(a: str, b: str) -> bool:
    return bool(a) and bool(b) and hmac.compare_digest(a, b)


def auth_master(authorization: str = Header(default="")) -> None:
    """Your bots. Never a browser."""
    if MASTER and authorization != f"Bearer {MASTER}":
        raise HTTPException(401, "bad master token")


def auth_control(authorization: str = Header(default="")) -> None:
    """The admin panel. Read + control, no /ingest."""
    if CONTROL and authorization != f"Bearer {CONTROL}":
        raise HTTPException(401, "bad control token")
```

```python
def _limit(key: str, n: int = 30, window: int = 60) -> None:
    now = time.time()
    q = _hits[key]
    while q and now - q[0] > window:
        q.popleft()
    if len(q) >= n:
        raise HTTPException(429, "slow down")
    q.append(now)
```

## Reach it from Render without Cloudflare

Render is public HTTPS, so it needs a public HTTPS API. Tailscale Funnel gives you exactly that, no Cloudflare:

```bash
tailscale up
tailscale funnel --bg 8787
# → https://your-pc.tailXXXX.ts.net
```

Funnel is public, so the control token matters. If you'd rather stay tailnet-only, use `tailscale serve --bg 8787` and open the panel from a device on your tailnet — but then Render can't reach it.

```html
      API = v.replace(/\/+$/, ""); localStorage.setItem("evie_api", API);
    }
    TOKEN = $("gateInput").value.trim();
    localStorage.setItem("evie_token", TOKEN);
    if (await ping()) { $("gate").classList.remove("on"); boot(); }
    else $("gateErr").textContent = "rejected";
  };
  $("gateInput").onkeydown = e => { if (e.key === "Enter") $("gateBtn").click(); };
  $("lock").onclick = () => { stop(); localStorage.removeItem("evie_token"); needConfig(); };
  $("cfg").onclick = () => {
    const v = prompt("API base URL", API);
    if (v) { localStorage.setItem("evie_api", v.replace(/\/+$/, "")); location.reload(); }
  };

  const H = () => ({ Authorization: `Bearer ${TOKEN}` });
  const q = () => `token=${encodeURIComponent(TOKEN)}`;

  async function ping() {
    if (!API) return false;
    try { return (await fetch(`${API}/health`, { headers: H() })).ok; }
    catch { return false; }
  }

  function say(msg) {
    $("actlog").textContent = `${new Date().toLocaleTimeString()}  ${msg}\n` + $("actlog").textContent;
  }

  // ---------- tabs ----------
  document.querySelectorAll(".tab").forEach(t => t.onclick = () => {
    document.querySelectorAll(".tab").forEach(x => x.classList.remove("active"));
    document.querySelectorAll(".pane").forEach(x => x.classList.remove("on"));
    t.classList.add("active");
    $(t.dataset.t).classList.add("on");
    if (t.dataset.t === "svc") loadServices();
  });

  // ---------- stats ----------
  async function loadStats() {
    try {
      const r = await fetch(`${API}/stats`, { headers: H() });
      if (!r.ok) return;
      const j = await r.json();
      $("sMembers").textContent = j.members.toLocaleString();
      $("sMsgs").textContent = j.messages.toLocaleString();
      $("sSpend").textContent = "$" + j.spend_usd.toLocaleString();
      $("sEvents").textContent = j.spend_events.toLocaleString() + " events";
      $("pauseBtn").textContent = j.paused ? "Resume ingest" : "Pause ingest";
    } catch {}
  }

  // ---------- live feed ----------
  function startSSE() {
    if (es) es.close();
    $("conn").textContent = "connecting";
    es = new EventSource(`${API}/live?${q()}`);
    es.onopen = () => { sseFails = 0; $("conn").textContent = "live"; $("dot").className = "dot on"; };
    es.onerror = () => {
      $("dot").className = "dot bad";
      $("conn").textContent = "reconnecting";
      if (++sseFails > 3) { es.close(); startPolling(); }
    };
    es.onmessage = e => { try { push(JSON.parse(e.data)); } catch {} };
  }
  let pollTimer = null, cursor = 0;
  function startPolling() {
    if (pollTimer) return;
    $("conn").textContent = "polling";
    pollTimer = setInterval(async () => {
      try {
        const r = await fetch(`${API}/feed?since=${cursor}&limit=100&${q()}`, { headers: H() });
        if (!r.ok) return;
        const j = await r.json();
        cursor = j.cursor; j.items.forEach(push);
        $("dot").className = "dot on";
      } catch { $("dot").className = "dot bad"; }
    }, 3000);
  }

  const seen = new Set();
  function push(ev) {
    if (!ev || seen.has(ev.pk)) return;
    seen.add(ev.pk); if (ev.pk > cursor) cursor = ev.pk;
    feedItems.push(ev);
    if (feedItems.length > 400) feedItems.shift();
    if (!paused) renderFeed();
  }
  function renderFeed() {
    const frag = document.createDocumentFragment();
    for (let i = feedItems.length - 1; i >= 0; i--) {
      const ev = feedItems[i];
      const hay = `${ev.text} ${ev.username || ""} ${ev.name || ""}`.toLowerCase();
      if (filter && !hay.includes(filter)) continue;
      const li = document.createElement("li");
      const money = /[$£€₦₹]\s?[\d,.]+/.test(ev.text || "");
      li.innerHTML =
        `<div class="dim mono" style="font-size:10px">${new Date(ev.ts*1000).toLocaleTimeString()} · chat ${ev.chat_id}</div>` +
        `<div class="who">@${ev.username || ev.sender_id} <span class="dim">${ev.name || ""}</span></div>` +
        `<div>${(ev.text || "").replace(/[&<>]/g, c => ({"&":"&amp;","<":"&lt;",">":"&gt;"}[c]))}</div>` +
        (money ? '<span class="pill">spend</span>' : "");
      frag.appendChild(li);
    }
    $("feed").replaceChildren(frag);
  }
  $("pauseFeed").onclick = e => { paused = !paused; e.target.textContent = paused ? "resume" : "pause"; if (!paused) renderFeed(); };
  $("clearFeed").onclick = () => { feedItems = []; seen.clear(); $("feed").replaceChildren(); };
  $("filter").oninput = e => { filter = e.target.value.toLowerCase(); renderFeed(); };

  // ---------- services ----------
  async function loadServices() {
    const r = await fetch(`${API}/control/services`, { headers: H() });
    if (!r.ok) { say("services: " + r.status); return; }
    const j = await r.json();
    const up = j.services.filter(s => s.state === "running").length;
    $("sSvc").textContent = `${up}/${j.services.length}`;
    $("sSvcNote").textContent = up === j.services.length ? "all healthy" : "degraded";

    $("svcList").replaceChildren(...j.services.map(s => {
      const row = document.createElement("div");
      row.className = "svc";
      const running = s.state === "running";
      row.innerHTML =
        `<div><b>${s.service}</b><div class="dim mono" style="font-size:11px">${s.container}</div></div>
         <span class="pill ${running ? "ok" : "warn"}">${s.state}</span>
         <div class="acts">
           <button data-s="${s.service}" data-v="restart">restart</button>
           <button data-s="${s.service}" data-v="${running ? "stop" : "start"}">${running ? "stop" : "start"}</button>
           <button data-s="${s.service}" data-v="logs">logs</button>
         </div>`;
      return row;
    }));

    $("svcList").querySelectorAll("button").forEach(b => b.onclick = async () => {
      const s = b.dataset.s, v = b.dataset.v;
      if (v === "logs") return loadLogs(s);
      b.disabled = true;
      say(`POST /control/${s}/${v}`);
      const r = await fetch(`${API}/control/${s}/${v}`, { method: "POST", headers: H() });
      say(`${s}/${v} → ${r.status}`);
      b.disabled = false;
      setTimeout(loadServices, 1200);
    });
  }

  async function loadLogs(service) {
    $("svcLog").textContent = "loading…";
    const r = await fetch(`${API}/control/logs/${service}?tail=200`, { headers: H() });
    const j = await r.json();
    $("svcLog").textContent = j.logs || "(empty)";
  }

  // ---------- quick actions ----------
  document.querySelectorAll(".qa button[data-act]").forEach(b => b.onclick = async () => {
    const [verb, svc] = b.dataset.act.split(":");
    b.disabled = true;
    if (verb === "recompute") {
      say("POST /control/recompute");
      const r = await fetch(`${API}/control/recompute`, { method: "POST", headers: H() });
      say(`recompute → ${r.status} ${JSON.stringify(await r.json().catch(() => ({})))}`);
    } else {
      say(`POST /control/${svc}/${verb}`);
      const r = await fetch(`${API}/control/${svc}/${verb}`, { method: "POST", headers: H() });
      say(`${svc}/${verb} → ${r.status}`);
    }
    b.disabled = false;
  });
  $("pauseBtn").onclick = async () => {
    const on = $("pauseBtn").textContent.startsWith("Pause");
    await fetch(`${API}/control/pause`, {
      method: "POST", headers: { ...H(), "Content-Type": "application/json" },
      body: JSON.stringify({ on }),
    });
    say(on ? "ingest paused" : "ingest resumed");
    loadStats();
  };
  $("refresh").onclick = () => { loadStats(); say("refreshed"); };

  // ---------- members ----------
  $("memLoad").onclick = async () => {
    const min = parseFloat($("memQ").value) || 0;
    const r = await fetch(`${API}/segment?min_ltv=${min}&limit=200`, { headers: H() });
    if (!r.ok) return;
    const j = await r.json();
    $("memBody").replaceChildren(...j.users.map(u => {
      const tr = document.createElement("tr");
      const risk = u.churn_risk || "—";
      tr.innerHTML =
        `<td class="mono">${u.telegram_id}</td>
         <td><span class="pill">${u.tier || "NEW"}</span></td>
         <td>$${(u.ltv_usd || 0).toFixed(2)}</td>
         <td>${u.order_count || 0}</td>
         <td><span class="pill ${risk === "HIGH" ? "hi" : risk === "LOW" ? "ok" : ""}">${risk}</span></td>`;
      return tr;
    }));
  };

  function stop() { if (es) es.close(); clearInterval(pollTimer); }

  function boot() {
    loadStats(); setInterval(loadStats, 15000);
    startSSE();
  }

  if (API && TOKEN) { ping().then(ok => ok ? boot() : needConfig()); }
  else needConfig();
})();
</script>
</body>
</html>
```

Then open your Render admin page, hit **config**, paste the funnel URL, paste the control token. You should get live messages within seconds.

## When it breaks

| Symptom | Cause |
|---|---|
| `permission denied /var/run/docker.sock` | `DOCKER_GID` wrong, or `data/` not chowned to 10001 |
| `AuthKeyDuplicatedError` | both sessions pointing at the same file |
| Panel says reconnecting forever | Funnel URL wrong, or CORS origin mismatch (`CC_ORIGIN` must match exactly, no trailing slash) |
| Cookie never sets | `COOKIE_SECURE=true` on an `http://` origin |
| Services list empty | `COMPOSE_PROJECT` doesn't match the actual compose project name |
| Logs show binary garbage | `_demux` needed — already handled, but only for `tty: false` containers |

Two things to know about the design: the panel talks to your PC over the public Funnel, so the **control token is the only thing between a stranger and your containers** — keep it long and rotate it if you ever paste it somewhere. And `pause` is a flag in SQLite, not a stopped container, so the listener keeps receiving and just stops writing — that's deliberate, since stopping the container loses `catch_up` coverage.
